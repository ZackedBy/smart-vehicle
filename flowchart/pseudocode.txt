BEGIN

CarInESPMode = True

WHILE CarInESPMode:

    IF TURN_Command_Given:
        IF Object_Blocking_Turn:
            Attempt_Turn_Opposite_Direction()
            IF Can_Complete_Turn:
                Complete_Turn()
		Stop()
            ELSE:
                Stop()
        ELSE:
            Complete_Turn()
	    Stop()

    IF MOVE_Command_Given:
        IF Object_Distance < 10cm:
            Display_Warning_Webpage()
            Car_Stops_And_Turns_90_Degrees()
            IF Path_Blocked:
                Stop()
            ELSE:
                Car_Drives_Until_Given_Distance()
        ELSE IF Object_Distance BETWEEN 10cm AND 30cm:
            Car_Drives_At_30_Percent_Speed()
        ELSE:
            Car_Drives_At_Full_Speed()
        
	Car_Drives_Until_Given_Distance()

END
